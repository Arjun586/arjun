    function setTheme(theme) {
      const html = document.documentElement;
      const lightBtn = document.getElementById('theme-light-btn');
      const darkBtn = document.getElementById('theme-dark-btn');

      if (theme === 'dark') {
        html.classList.add('dark');
        localStorage.setItem('theme', 'dark');
      } else {
        html.classList.remove('dark');
        localStorage.setItem('theme', 'light');
      }
    }

    // Initialize Theme on load
    (function initTheme() {
      const savedTheme = localStorage.getItem('theme');
      if (savedTheme === 'dark' || (!savedTheme && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
        // default to light mode to show off linen aesthetic first, or respect dark
        document.documentElement.classList.remove('dark');
      }
    })();

    function openCaseStudy(projectId) {
      const data = projectCaseStudies[projectId];
      if (!data) return;

      document.getElementById('modal-case-badge').innerText = data.badge;
      document.getElementById('modal-case-ref').innerText = data.ref;
      document.getElementById('modal-case-title').innerText = data.title;
      document.getElementById('modal-case-desc').innerText = data.desc;

      const modal = document.getElementById('case-study-modal');
      modal.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
    }

    function closeCaseStudy(e) {
      const modal = document.getElementById('case-study-modal');
      modal.classList.add('hidden');
      document.body.style.overflow = 'auto';
    }

    // Close modal on Escape key
    window.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeCaseStudy();
    });

    // 6. Navigation Active Spy for Left Sidebar
    window.addEventListener('scroll', () => {
      const sections = ['home', 'about', 'experience', 'projects', 'skills', 'contact'];
      const scrollPos = window.scrollY + 200;

      for (let i = sections.length - 1; i >= 0; i--) {
        const sec = document.getElementById(sections[i]);
        if (sec && sec.offsetTop <= scrollPos) {
          updateSidebarActive(sections[i]);
          break;
        }
      }
    });

    function updateSidebarActive(activeId) {
      const links = document.querySelectorAll('#sidebar-nav .nav-link');
      links.forEach(link => {
        const target = link.getAttribute('data-target');
        const indicator = link.querySelector('.indicator');
        if (target === activeId || (activeId === 'home' && target === 'home')) {
          link.classList.add('border-espresso-850', 'dark:border-[#EDEDEC]', 'bg-linen-200/60', 'dark:bg-espresso-800/60', 'text-espresso-850', 'dark:text-[#EDEDEC]', 'font-semibold');
          link.classList.remove('border-transparent', 'text-espresso-muted', 'dark:text-[#9A948D]');
          if (indicator) {
            indicator.classList.add('bg-espresso-850', 'dark:bg-[#EDEDEC]');
            indicator.classList.remove('bg-transparent');
          }
        } else {
          link.classList.remove('border-espresso-850', 'dark:border-[#EDEDEC]', 'bg-linen-200/60', 'dark:bg-espresso-800/60', 'text-espresso-850', 'dark:text-[#EDEDEC]', 'font-semibold');
          link.classList.add('border-transparent', 'text-espresso-muted', 'dark:text-[#9A948D]');
          if (indicator) {
            indicator.classList.remove('bg-espresso-850', 'dark:bg-[#EDEDEC]');
            indicator.classList.add('bg-transparent');
          }
        }
      });
    }
